async function loadU(user) {
  const ref  = db.collection('users').doc(user.uid);
  const snap = await ref.get();
  if (snap.exists) {
    U = snap.data();
  } else {
    U = defProfile(user);
    await ref.set(U);
  }
  const today = yyyymmdd();
  if (U.stepDate !== today) { U.steps = 0; U.stepDate = today; sched(); }
}

function authTab(tab) {
  // Hide all panels
  ['login','signup','forgot'].forEach(p => {
    const el = document.getElementById('fp-'+p);
    if (el) el.classList.remove('on');
  });
  // Show/hide the auth tabs row
  const tabRow = document.getElementById('auth-tabs');
  if (tab === 'forgot') {
    tabRow.style.display = 'none';
    document.getElementById('fp-forgot').classList.add('on');
    fclr('err-forgot','ok-forgot');
  } else {
    tabRow.style.display = 'flex';
    document.getElementById('fp-'+tab).classList.add('on');
    // Highlight active tab button
    tabRow.querySelectorAll('.tb').forEach((b,i) => {
      b.classList.toggle('on', (i===0&&tab==='login') || (i===1&&tab==='signup'));
    });
    fclr('err-login','err-signup','ok-signup');
  }
}

async function doLogin() {
  fclr('err-login');
  const email = document.getElementById('f-lemail').value.trim();
  const pass  = document.getElementById('f-lpass').value;
  if (!email||!pass) { ferr('err-login','Please fill in both fields.'); return; }
  setBtnLd('btn-login', true);
  try { await auth.signInWithEmailAndPassword(email, pass); }
  catch(e) { ferr('err-login', errMsg(e.code)); setBtnLd('btn-login', false); }
}

async function doSignup() {
  fclr('err-signup','ok-signup');
  const name  = document.getElementById('f-sname').value.trim();
  const email = document.getElementById('f-semail').value.trim();
  const pass  = document.getElementById('f-spass').value;
  const goal  = document.getElementById('f-sgoal').value  || 'general';
  const level = document.getElementById('f-slevel').value || 'beginner';
  if (!name)       { ferr('err-signup','Enter your name.'); return; }
  if (!email)      { ferr('err-signup','Enter your email.'); return; }
  if (pass.length<6){ ferr('err-signup','Password must be 6+ characters.'); return; }
  setBtnLd('btn-signup', true);
  try {
    const cred = await auth.createUserWithEmailAndPassword(email, pass);
    await cred.user.updateProfile({ displayName: name });
    await cred.user.sendEmailVerification();
    const p = { ...defProfile(cred.user), name, email, goal, fitnessLevel:level };
    await db.collection('users').doc(cred.user.uid).set(p);
    setBtnLd('btn-signup', false);
    // onAuthStateChanged fires next → routes to verify screen
  } catch(e) { ferr('err-signup', errMsg(e.code)); setBtnLd('btn-signup', false); }
}

async function doForgotPw() {
  fclr('err-forgot','ok-forgot');
  const email = document.getElementById('f-femail').value.trim();
  if (!email) { ferr('err-forgot','Enter your email.'); return; }
  setBtnLd('btn-forgot', true);

}

    // territory page uses flex, others use block
    if (p === 'territory') {
      pg.style.display = isActive ? 'flex' : 'none';
      pg.classList.toggle('on', isActive);
    } else {
      pg.style.display = isActive ? 'block' : 'none';
      pg.classList.toggle('on', isActive);
    }
    if (bni) {
      bni.classList.remove('on','walk-live');
      if (isActive) bni.classList.add('on');
      if (p === 'territory' && walkOn) bni.classList.add('walk-live');
    }
  });

  if (name === 'progress') refreshUI();

  // ── KEY FIX: init map only when territory page is visible ──
  if (name === 'territory') {
    if (!mapReady) {
      // Use requestAnimationFrame so DOM has painted the flex layout
      requestAnimationFrame(() => requestAnimationFrame(() => initMap()));
    } else {
      // Map exists but might have wrong size (was hidden)
      leafMap.invalidateSize();
    }
  }
}

/* ── MAP INIT ───────────────────────────────── */
function initMap() {
  const el = document.getElementById('tmap');
  if (!el || mapReady) return;

  // Force a layout read to make sure tmap has a real height
  const h = el.getBoundingClientRect().height;
  if (h < 50) {
    // Not painted yet — retry
    setTimeout(() => initMap(), 100);
    return;
  }

  leafMap = L.map('tmap', {
    center: [20.9, 73.1],  // default fallback centre (India)
    zoom: 15,
    zoomControl: true,
    attributionControl: false
  });

  L.tileLayer('https://{s}.basemaps.cartocdn.com/dark_all/{z}/{x}/{y}{r}.png', {
    maxZoom:19, subdomains:'abcd'
  }).addTo(leafMap);

  mapReady = true;

  // Try to jump to user's real location
  if (navigator.geolocation) {
    navigator.geolocation.getCurrentPosition(
      pos => leafMap.setView([pos.coords.latitude, pos.coords.longitude], 16),
      ()  => {},
      { timeout: 6000, maximumAge: 60000 }
    );
  }

  // Start listening to tiles now that map + auth are both ready
  subTiles();
}

/* ── FIRESTORE TILES (real-time) ────────────── */
function subTiles() {
  if (tileUnsub) return;                     // already subscribed
  if (!FBU) return;                          // not authed yet — shouldn't happen here

  tileUnsub = db.collection('tiles')
    .limit(1000)
    .onSnapshot(snap => {
      tileCache = [];
      snap.forEach(doc => {
        const t = doc.data();
        tileCache.push(t);
        renderTile(t);
      });
      // Remove rects for tiles that were deleted
      const live = new Set(tileCache.map(t => t.id));
      Object.keys(tileRects).forEach(id => {
        if (!live.has(id)) { tileRects[id].remove(); delete tileRects[id]; }
      });
      rebuildLeaderboard();
      rebuildMyPanel();
      updateHomeTerr();
    }, err => console.error('Tile snapshot error:', err));
}

function renderTile(tile) {
  if (!mapReady || !leafMap) return;
  const bounds = tileBounds(tile.id);
  const color  = tile.ownerColor || '#666';
  const isMe   = tile.ownerUID === FBU?.uid;
  const style  = { fillColor:color, fillOpacity:.42, color:isMe?'#fff':color, weight:isMe?2:1, opacity:.75 };

  if (tileRects[tile.id]) {
    tileRects[tile.id].setStyle(style);
    // Refresh popup content too
    tileRects[tile.id].getPopup()?.setContent(tilePopupHTML(tile, isMe));
  } else {
    const rect = L.rectangle(bounds, {...style, interactive:true}).addTo(leafMap);
    rect.bindPopup(tilePopupHTML(tile, isMe), {maxWidth:220});
    tileRects[tile.id] = rect;
  }
}

function tilePopupHTML(tile, isMe) {
  const date = tile.capturedAt?.toDate ? tile.capturedAt.toDate().toLocaleDateString() : '—';
  const km   = ((tile.captureM||0)/1000).toFixed(2);
  return `<div class="tile-pop-name" style="color:${tile.ownerColor||'#fff'}">${tile.ownerName}'s Tile</div>
<div style="font-size:.7rem;color:var(--muted)">Captured ${date}</div>
<div class="tile-pop-row"><span>Defender walked:</span><span style="color:${tile.ownerColor||'#fff'};font-weight:700">${km} km</span></div>
<div class="tile-pop-row" style="margin-top:.2rem">${
  isMe
    ? `<span style="color:var(--a2)">🛡️ Your tile</span>`
    : `<span>To conquer, walk</span><span style="color:var(--a3);font-weight:700">&gt;${Math.ceil(tile.captureM||0)}m here</span>`
}</div>`;
}

/* ── TILE MATH ──────────────────────────────── */
function tileId(lat, lng) {
  return `${Math.floor(lat/TILE_SIZE)}_${Math.floor(lng/TILE_SIZE)}`;
}
function tileBounds(id) {
  const [r,c] = id.split('_').map(Number);
  return [[r*TILE_SIZE, c*TILE_SIZE], [(r+1)*TILE_SIZE, (c+1)*TILE_SIZE]];
}
function tileCenter(id) {
  const [r,c] = id.split('_').map(Number);
  return [(r+.5)*TILE_SIZE, (c+.5)*TILE_SIZE];
}
function haversine(la1,lo1,la2,lo2) {
  const R=6371000, f1=la1*Math.PI/180, f2=la2*Math.PI/180,
        df=(la2-la1)*Math.PI/180, dl=(lo2-lo1)*Math.PI/180;
  const a=Math.sin(df/2)**2+Math.cos(f1)*Math.cos(f2)*Math.sin(dl/2)**2;
  return R*2*Math.atan2(Math.sqrt(a),Math.sqrt(1-a));
}

/* ── WALK: GPS CALLBACK ─────────────────────── */
function onGPS(pos) {
  if (!walkOn) return;
  hideGpsErr();

  const { latitude:lat, longitude:lng, accuracy } = pos.coords;
  if (accuracy > 55) return;  // ignore inaccurate fixes

  const ts = pos.timestamp;

  // Move markers
  walkDot.setLatLng([lat, lng]);
  walkLine.addLatLng([lat, lng]);
  leafMap.panTo([lat, lng], {animate:true, duration:.4});

  if (walkLastPos) {
    const dist = haversine(walkLastPos.lat, walkLastPos.lng, lat, lng);
    const dt   = (ts - walkLastPos.ts) / 1000;

    if (dt > 0 && dt < 60) {
      const spd = dist / dt;  // m/s
      if (spd <= MAX_WALK_MS && dist > 0.5) {
        walkMeters += dist;

        const tid = tileId(lat, lng);
        walkTileDist[tid] = (walkTileDist[tid] || 0) + dist;

        // Temp highlight this tile
        if (!visitTmpRects[tid] && !tileRects[tid]) {
          const b = tileBounds(tid);
          visitTmpRects[tid] = L.rectangle(b, {
            fillColor:myColor, fillOpacity:.15, color:myColor, weight:1, opacity:.4, interactive:false
          }).addTo(leafMap);
        }

        // Update HUD
        document.getElementById('hud-spd').textContent = (spd*3.6).toFixed(1);
      }
    }
  }

  walkLastPos = { lat, lng, ts };

  // Update HUD counters
  document.getElementById('hud-m').textContent     = Math.round(walkMeters);
  document.getElementById('hud-tiles').textContent = Object.keys(walkTileDist).length;
  document.getElementById('hud-cap').textContent   = Object.values(walkTileDist).filter(d=>d>=MIN_CAP_M).length;
}

/* ── CAPTURE LOGIC ──────────────────────────── */
async function doCaptures() {
  const eligible = Object.entries(walkTileDist).filter(([,d]) => d >= MIN_CAP_M);
  const res = { newTiles:0, conquered:0, defended:0, events:[] };

  for (const [tid, myM] of eligible) {
    try {
      const ref = db.collection('tiles').doc(tid);
      await db.runTransaction(async tx => {
        const snap = await tx.get(ref);
        const old  = snap.exists ? snap.data() : null;
        const [clat, clng] = tileCenter(tid);

        if (!old) {
          // Unclaimed → claim it
          tx.set(ref, {
            id: tid,
            tileLat: clat, tileLng: clng,
            ownerUID:   FBU.uid,
            ownerName:  U.name,
            ownerColor: myColor,
            captureM:   myM,
            capturedAt: firebase.firestore.FieldValue.serverTimestamp(),
            prevOwnerUID:  null,
            prevOwnerName: null
          });
          res.newTiles++;
          res.events.push({ k:'new', myM });
        } else if (old.ownerUID === FBU.uid) {
          // Already mine — update if improved
          if (myM > (old.captureM||0)) {
            tx.update(ref, { captureM: myM });
          }
        } else {
          // Owned by someone else — must beat their distance
          if (myM > (old.captureM||0)) {
            tx.update(ref, {
              ownerUID:   FBU.uid,
              ownerName:  U.name,
              ownerColor: myColor,
              captureM:   myM,
              capturedAt: firebase.firestore.FieldValue.serverTimestamp(),
              prevOwnerUID:  old.ownerUID,
              prevOwnerName: old.ownerName
            });
            res.conquered++;
            res.events.push({ k:'conquered', myM, prev:old.ownerName, prevM:old.captureM||0 });
          } else {
            res.defended++;
            res.events.push({ k:'defended', myM, owner:old.ownerName, ownerM:old.captureM||0 });
          }
        }
      });
    } catch(e) { console.error('Capture tx error', tid, e); }
  }
  return res;
}

function initSteps() {
  const st=document.getElementById('step-status');
  updateStepUI(U.steps||0);
  if (window.DeviceMotionEvent) {
    let hist=[],last=0;
    const handle=e=>{
      const a=e.accelerationIncludingGravity;
      if (!a||a.x===null) return;
      const mag=Math.sqrt(a.x**2+a.y**2+a.z**2);
      hist.push(mag); if(hist.length>20)hist.shift();
      const avg=hist.reduce((a,b)=>a+b,0)/hist.length;
      const now=Date.now();
      if (Math.abs(mag-avg)>12&&now-last>300) {
        last=now; if(!U)return;
        U.steps=(U.steps||0)+1; updateStepUI(U.steps); sched(); st.textContent='Counting...';
      }
    };
    if (typeof DeviceMotionEvent.requestPermission==='function') {
      st.textContent='Tap to enable'; st.style.cursor='pointer';
      st.onclick=()=>DeviceMotionEvent.requestPermission().then(s=>{

/* ── UI REFRESH ─────────────────────────────── */
function refreshUI() {
  if (!U) return;
  document.getElementById('s-streak').textContent  = U.streak+'🔥';
  document.getElementById('s-wk').textContent      = U.workouts;
  document.getElementById('s-xp').textContent      = U.points;
  document.getElementById('hero-name').innerHTML   = (U.name||'Champion').toUpperCase().split(' ')[0]+' <span>💪</span>';

  let li=0;
  for(let i=LEVELS.length-1;i>=0;i--){if(U.points>=LEVELS[i].x){li=i;break;}}
  const nx=LEVELS[li+1]?LEVELS[li+1].x:LEVELS[li].x, px=LEVELS[li].x;
  const p=li+1<LEVELS.length?Math.round(((U.points-px)/(nx-px))*100):100;
  document.getElementById('lvl-name').textContent='Level '+(li+1)+' — '+LEVELS[li].n;
  document.getElementById('lvl-xp').textContent=U.points+' / '+nx+' XP';
  document.getElementById('lvl-fill').style.width=p+'%';

  document.getElementById('pg-wk').textContent  = U.workouts;
  document.getElementById('pg-str').textContent = '🔥 '+U.streak+' day streak';
  document.getElementById('pg-bst').textContent = U.bestStreak;
  document.getElementById('pg-rep').textContent = U.totalReps;
  document.getElementById('pg-cal').textContent = Math.round(U.calories);

  buildWeekChart(); buildBreakdown(); buildBadges();
  updateHomeTerr(); updateStepUI(U.steps||0);
}

function buildWeekChart() {
  const days=['Mo','Tu','We','Th','Fr','Sa','Su'], ti=(new Date().getDay()+6)%7;
  const data=U.weeklyData||[0,0,0,0,0,0,0], mx=Math.max(...data,1);
  document.getElementById('week-bars').innerHTML=days.map((d,i)=>{
    const h=Math.round((data[i]/mx)*62);
    return `<div class="bw"><div class="bbar${i===ti?' td':''}" style="height:${h}px"></div><div class="blbl">${d}</div></div>`;
  }).join('');
}

function buildBreakdown() {
  const ec=U.exerciseCounts||{};
  const el=document.getElementById('ex-break');
  if(!Object.keys(ec).length){el.innerHTML='<p style="font-size:.76rem;color:var(--muted)">No exercises yet.</p>';return;}
  el.innerHTML=Object.entries(ec).sort((a,b)=>b[1]-a[1]).map(([id,reps])=>{
    const ex=EXERCISES.find(e=>e.id===id);
    return `<div style="display:flex;justify-content:space-between;align-items:center;padding:.55rem .8rem;background:var(--card);border:1px solid var(--border);border-radius:9px;margin-bottom:.4rem;">
      <span>${ex?ex.ic:''} ${ex?ex.n:id}</span>
      <span style="font-family:var(--fm);font-size:.76rem;color:var(--a2)">${reps} reps</span></div>`;
  }).join('');
}

function buildBadges() {
  const html=BADGES.map(b=>{
    const ok=b.r(U);
    return `<div class="bdg ${ok?'ok':'locked'}"><div class="bi">${b.ic}</div><div class="bn">${b.n}</div><div class="bd">${b.d}</div></div>`;
  }).join('');
  ['badges-home','badges-prog'].forEach(id=>{const e=document.getElementById(id);if(e)e.innerHTML=html;});
}

  document.querySelectorAll(grp+' .pl-opt').forEach(o=>{

    if(oc.includes("'"+val+"'")) o.classList.add('on'); else o.classList.remove('on');
  });
}

function yyyymmdd(offsetDays=0) {
  const d=new Date(Date.now()+offsetDays*86400000);
  return d.toISOString().split('T')[0];
}

function uidColor(uid) {
  let h=5381;
  for(let i=0;i<uid.length;i++) h=((h<<5)+h)+uid.charCodeAt(i);
  return PLAYER_COLORS[Math.abs(h)%PLAYER_COLORS.length];
}
